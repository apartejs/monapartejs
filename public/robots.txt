# /debug/ exposes the last exchange sent to the model — a debugging tool,
# of no interest to a search engine, and one we don't want showing up in
# search results.
#
# /chat/ is deliberately NOT disallowed here: index.html's canonical tag is
# enough to consolidate these addresses on the root, and a disallow would
# also stop social crawlers from reading the card for a shared link. The
# conversations only exist in the visitor's browser: these addresses carry
# no data.
User-agent: *
Allow: /
Disallow: /debug/
Disallow: /unregister-sw.html
